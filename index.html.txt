  <div id="player-container">
    <h2 id="current-game-title">Game Title</h2>
    <iframe id="game-frame" src="" allowfullscreen></iframe>
    <div class="controls">
      <button onclick="closeGame()">Close Game</button>
      <button onclick="goFullscreen()">Fullscreen</button>
    </div>
  </div>

  <!-- Game Grid -->
  <div class="grid" id="game-grid"></div>

  <script>
    // Add your games here
    const games = [
      {
        title: "2048",
        url: "https://play2048.co"
      },
      {
        title: "Scratch Example",
        url: "https://mit.edu"
      }
      // Copy and paste the block above to add more games
    ];

    const grid = document.getElementById("game-grid");
    const playerContainer = document.getElementById("player-container");
    const gameFrame = document.getElementById("game-frame");
    const gameTitle = document.getElementById("current-game-title");

    function renderGames(list) {
      grid.innerHTML = "";
      list.forEach(game => {
        const card = document.createElement("div");
        card.className = "card";
        card.innerHTML = `<h3>${game.title}</h3>`;
        card.onclick = () => launchGame(game);
        grid.appendChild(card);
      });
    }

    function launchGame(game) {
      gameTitle.innerText = game.title;
      gameFrame.src = game.url;
      playerContainer.style.display = "block";
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function closeGame() {
      gameFrame.src = "";
      playerContainer.style.display = "none";
    }

    function goFullscreen() {
      if (gameFrame.requestFullscreen) {
        gameFrame.requestFullscreen();
      }
    }

    function filterGames() {
      const query = document.getElementById("search").value.toLowerCase();
      const filtered = games.filter(g => g.title.toLowerCase().includes(query));
      renderGames(filtered);
    }

    // Load all games on startup
    renderGames(games);
  </script>
